+const opAmend = "bulk-amend.v8"
+
+// An amendment may raise an amount by at most maxIncreasePercent and never moves it across the
+// second-approval line (secondApprovalOver in approval.go).
+const maxIncreasePercent = 10
+
+// newAmend serves the requester-side correction of the amounts of their own submitted purchases.
+func newAmend(pool *pgxpool.Pool) protocol.AmendHandler {
+	return func(ctx context.Context, u app.Identity, key string, ids []string, amounts map[string]int, reason string) (protocol.AmendResult, bool, error) {
+		for _, id := range ids {
+			if strings.ContainsRune(id, 0) {
+				return protocol.AmendResult{}, false, &protocol.Rejection{Status: 404}
+			}
+		}
+		tx, err := pool.Begin(ctx)
+		if err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		defer tx.Rollback(ctx)
+		if _, err = tx.Exec(ctx, `SELECT set_config('app.tenant',$1,true),set_config('app.user',$2,true),set_config('app.role',$3,true)`, u.Tenant, u.User, u.Role); err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		lock, _ := json.Marshal([]string{opAmend, u.Tenant, u.User, key})
+		if _, err = tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1,0))`, string(lock)); err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		entries := make([]string, len(ids))
+		for i, id := range ids {
+			entries[i] = fmt.Sprintf("%s=%d", id, amounts[id])
+		}
+		hash := ledger.Fingerprint(opAmend, entries)
+		var savedHash string
+		var saved []byte
+		err = tx.QueryRow(ctx, `SELECT "requestHash",response FROM "ActionRequest" WHERE "tenantId"=$1 AND "actorId"=$2 AND operation=$3 AND key=$4`,
+			u.Tenant, u.User, opAmend, key).Scan(&savedHash, &saved)
+		switch {
+		case err == nil && savedHash != hash:
+			return protocol.AmendResult{}, false, &protocol.Rejection{Status: 409}
+		case err == nil:
+			var body map[string][]protocol.AmendItem
+			if err = json.Unmarshal(saved, &body); err != nil {
+				return protocol.AmendResult{}, false, err
+			}
+			return protocol.AmendResult{Field: "amended", Items: body["amended"]}, true, tx.Commit(ctx)
+		case !errors.Is(err, pgx.ErrNoRows):
+			return protocol.AmendResult{}, false, err
+		}
+		rows, err := tx.Query(ctx, `SELECT id,amount FROM "Purchase" WHERE id=ANY($1::text[]) AND "tenantId"=$2 AND "createdBy"=$3 AND status='submitted' FOR UPDATE`,
+			ids, u.Tenant, u.User)
+		if err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		old := map[string]int{}
+		for rows.Next() {
+			var id string
+			var amount int
+			if err = rows.Scan(&id, &amount); err != nil {
+				rows.Close()
+				return protocol.AmendResult{}, false, err
+			}
+			old[id] = amount
+		}
+		err = rows.Err()
+		rows.Close()
+		if err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		// An ineligible target and an unacceptable amount are refused alike, for the whole batch.
+		items := make([]protocol.AmendItem, len(ids))
+		batch := &pgx.Batch{}
+		for i, id := range ids {
+			from, ok := old[id]
+			to := amounts[id]
+			if !ok || int64(to)*100 > int64(from)*(100+maxIncreasePercent) || (from > secondApprovalOver) != (to > secondApprovalOver) {
+				return protocol.AmendResult{}, false, &protocol.Rejection{Status: 409}
+			}
+			items[i] = protocol.AmendItem{ID: id, Amount: to}
+			batch.Queue(`UPDATE "Purchase" SET amount=$4 WHERE id=$1 AND "tenantId"=$2 AND "createdBy"=$3 AND status='submitted'`, id, u.Tenant, u.User, to)
+			batch.Queue(`INSERT INTO "AmendmentHistory" ("tenantId","purchaseId","fromAmount","toAmount","changedBy",operation,"idempotencyKey",reason) VALUES ($1,$2,$3,$4,$5,$6,$7,$8)`,
+				u.Tenant, id, from, to, u.User, opAmend, key, reason)
+			batch.Queue(`INSERT INTO "PurchaseEvent" ("tenantId","purchaseId",type,"actorId","idempotencyKey") VALUES ($1,$2,'purchase.amended',$3,$4)`,
+				u.Tenant, id, u.User, key)
+		}
+		result := protocol.AmendResult{Field: "amended", Items: items}
+		response, err := json.Marshal(result)
+		if err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		batch.Queue(`INSERT INTO "ActionRequest" ("tenantId","actorId",operation,key,"requestHash",response) VALUES ($1,$2,$3,$4,$5,$6::jsonb)`,
+			u.Tenant, u.User, opAmend, key, hash, string(response))
+		if err = tx.SendBatch(ctx, batch).Close(); err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		if err = tx.Commit(ctx); err != nil {
+			return protocol.AmendResult{}, false, err
+		}
+		return result, false, nil
+	}
+}
