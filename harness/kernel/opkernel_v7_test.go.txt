// v7 private regression (copied into the overlaid public tree by harness/kernel_regression.py):
// under a linguistic collation ORDER BY p.id differs from the Go byte order of ids ("B2" < "a1" in bytes,
// "a1" < "B2" in en-US). The private kernel's two managed columns must both pair every old value with its
// own target in the column's history table.
func TestV7FieldUpdatesPairOldValuesByIDUnderLinguisticCollation(t *testing.T) {
	d := open(t)
	ctx := context.Background()
	var icu bool
	if err := d.admin.QueryRow(ctx, `SELECT EXISTS (SELECT 1 FROM pg_collation WHERE collname='en-US-x-icu')`).Scan(&icu); err != nil || !icu {
		t.Fatal("ICU collation en-US-x-icu required for the v7 regression") // never a skip
	}
	for _, sql := range []string{
		`DROP TABLE IF EXISTS "CostCenterHistory","CostCenter" CASCADE`,
		`ALTER TABLE "Purchase" ALTER COLUMN id TYPE text COLLATE "en-US-x-icu"`,
		read(t, "tasks/cost-center/schema.sql"),
	} {
		if _, err := d.admin.Exec(ctx, sql); err != nil {
			t.Fatal(err)
		}
	}
	d.purchase("a1", "alice", 9, "submitted")
	d.purchase("B2", "alice", 9, "submitted")
	if _, err := d.admin.Exec(ctx, `UPDATE "Purchase" SET "assigneeId"=CASE id WHEN 'a1' THEN 'carol' ELSE 'hana' END,
	 "costCenterId"=CASE id WHEN 'a1' THEN 'CC-OPS' ELSE 'CC-RND' END WHERE id IN ('a1','B2')`); err != nil {
		t.Fatal(err)
	}
	if got := d.rows(`SELECT id FROM "Purchase" WHERE id IN ('a1','B2') ORDER BY id`); got[0] != "a1" {
		t.Fatalf("collation not linguistic: %v", got)
	}
	if _, _, err := d.exec.Run(ctx, bob, opdef.BulkReassign, "kr", Request{IDs: []string{"a1", "B2"}, Value: "carol"}); err != nil {
		t.Fatal(err)
	}
	d.want("assignment history pairs old value with its own target",
		d.rows(`SELECT "purchaseId","fromAssignee","toAssignee" FROM "AssignmentHistory" WHERE "idempotencyKey"='kr' ORDER BY "purchaseId" COLLATE "C"`),
		"B2 hana carol", "a1 carol carol")
	if _, _, err := d.exec.Run(ctx, bob, opdef.CostCenter, "kc", Request{IDs: []string{"a1", "B2"}, Value: "CC-OPS"}); err != nil {
		t.Fatal(err)
	}
	d.want("cost-center history pairs old value with its own target",
		d.rows(`SELECT "purchaseId","fromCostCenter","toCostCenter" FROM "CostCenterHistory" WHERE "idempotencyKey"='kc' ORDER BY "purchaseId" COLLATE "C"`),
		"B2 CC-RND CC-OPS", "a1 CC-OPS CC-OPS")
}
