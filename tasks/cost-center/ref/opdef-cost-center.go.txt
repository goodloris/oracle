package candidate

import (
	"context"

	"github.com/XUJiahua/enterprise_agent/internal/app"
	"github.com/XUJiahua/enterprise_agent/internal/opdef"
	"github.com/XUJiahua/enterprise_agent/internal/opkernel"
	"github.com/XUJiahua/enterprise_agent/internal/protocol"
)

func costCenter(exec *opkernel.Executor) protocol.ParamHandler {
	op := entry(opdef.CostCenter)
	return func(ctx context.Context, u app.Identity, key string, ids []string, value string) (protocol.ParamResult, bool, error) {
		done, replayed, err := exec.Run(ctx, u, opdef.CostCenter, key, opkernel.Request{IDs: ids, Value: value})
		if err != nil {
			return protocol.ParamResult{}, false, adapt(err)
		}
		return protocol.ParamResult{Field: op.Response, IDs: done, Name: op.Param, Value: value}, replayed, nil
	}
}
