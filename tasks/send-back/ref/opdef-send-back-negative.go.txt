package candidate

import (
	"context"

	"github.com/XUJiahua/enterprise_agent/internal/app"
	"github.com/XUJiahua/enterprise_agent/internal/opdef"
	"github.com/XUJiahua/enterprise_agent/internal/opkernel"
	"github.com/XUJiahua/enterprise_agent/internal/protocol"
)

// Explicit negative control: a hand-made adapter that reports the entry's From state instead of To.
func sendBack(exec *opkernel.Executor) protocol.ItemHandler {
	op := entry(opdef.SendBack)
	return func(ctx context.Context, u app.Identity, key, id, reason string) (protocol.ItemResult, bool, error) {
		done, replayed, err := exec.Run(ctx, u, opdef.SendBack, key, opkernel.Request{IDs: []string{id}, Reason: reason})
		if err != nil {
			return protocol.ItemResult{}, false, adapt(err)
		}
		return protocol.ItemResult{ID: done[0], Status: op.From}, replayed, nil
	}
}
